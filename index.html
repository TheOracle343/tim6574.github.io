========================================== # LOGIC FUNCTIONS # ========================================== def generate_terrain(): for x in range(MAP_W): for y in range(LOGICAL_H): terrain_bmp[x, y] = 0 offset1 = random.uniform(20.0, 50.0) offset2 = random.uniform(30.0, 70.0) base_h = random.randint(80, 120) for x in range(MAP_W): h = base_h + int(math.sin(x/offset1)*20 + math.cos(x/offset2)*30) if h < 40: h = 40 if h > 150: h = 150 for y in range(h, LOGICAL_H): terrain_bmp[x, y] = 1 def settle_tank(tx, ty): while ty < LOGICAL_H - TANK_H: if terrain_bmp[int(tx + TANK_W/2), int(ty + TANK_H)] == 1: break ty += 1 return ty def carve_crater(cx, cy, r): for dx in range(-r, r+1): for dy in range(-r, r+1): if dx*dx + dy*dy <= r*r: nx, ny = int(cx + dx), int(cy + dy) if 0 <= nx < MAP_W and 0 <= ny < LOGICAL_H: terrain_bmp[nx, ny] = 0 def get_barrel_pts(angle, length=24): cx, cy = TANK_W // 2, -2 ex, ey = cx + int(math.cos(angle) * length), cy - int(math.sin(angle) * length) # Doubled barrel thickness width (-2 to +2 instead of -1 to +1) return [(cx, cy - 2), (ex, ey), (cx, cy + 2)] def calc_damage(tx, ty, cx, cy): dist = math.sqrt((tx+TANK_W/2 - cx)**2 + (ty+TANK_H/2 - cy)**2) if dist < exp_radius + TANK_W: return int((exp_radius + TANK_W - dist) * 2.0) return 0 #
